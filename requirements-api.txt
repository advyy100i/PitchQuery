# Serving dependencies only — what a hosted API needs and nothing more.
#
# Deliberately absent: torch and sentence-transformers (~2.5 GB installed, and
# more resident memory than a free tier has). The API detects their absence and
# runs sparse + shape + planner, reporting `"rankers": "sparse only"` on /health
# rather than pretending it is still fusing two rankers.
#
# Also absent: pandas. The xG model used to be a pickle fitted on a pandas
# frame, so serving one shot meant rebuilding that frame — ~70 MB resident for
# twelve numbers. models/export_xg_portable.py re-serialises the model to
# LightGBM's own text format plus a two-parameter sigmoid, so scoring is numpy
# and nothing else.
#
# Local development uses requirements.txt + requirements-ml.txt instead.

python-dotenv>=1.0

# Retrieval: the TF-IDF index is fitted from the database at startup.

lightgbm>=4.0
